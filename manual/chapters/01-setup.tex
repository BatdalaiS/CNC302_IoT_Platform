ThingsBoard + Postgres нь эхлэх үедээ санах ой ихээр шаарддаг. Анхны тохиргоогоор Pi-гийн swap ердөө 200 MB --- энэ нь Лаб 1-д "container killed (OOM)" алдаа өгнө.

\begin{Shaded}
\begin{Highlighting}[]
\FunctionTok{sudo}\NormalTok{ dphys{-}swapfile swapoff}
\FunctionTok{sudo}\NormalTok{ sed }\AttributeTok{{-}i} \StringTok{\textquotesingle{}s/\^{}CONF\_SWAPSIZE=.*/CONF\_SWAPSIZE=2048/\textquotesingle{}}\NormalTok{ /etc/dphys{-}swapfile}

\FunctionTok{sudo}\NormalTok{ dphys{-}swapfile swapon}
\FunctionTok{free} \AttributeTok{{-}h}                            \CommentTok{\# Swap: 2.0Gi байх ёстой}
\end{Highlighting}
\end{Shaded}

\section{В. VS Code Remote-SSH холболт}\label{ux432.-vs-code-remote-ssh-ux445ux43eux43bux431ux43eux43bux442}}

\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  VS Code → \texttt{Ctrl+Shift+P} → \textbf{Remote-SSH: Add New SSH Host}
\item
  \texttt{ssh\ -i\ \textasciitilde{}/.ssh/cnc302\ cnc302@pi-team03.local}
\item
  Холбогдоод \texttt{\textasciitilde{}/cnc302} фолдерыг нээнэ
